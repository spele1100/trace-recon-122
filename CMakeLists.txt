cmake_minimum_required(VERSION 3.16)
project(trace_recon_122 CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(Eigen3 REQUIRED)

find_package(pybind11 CONFIG QUIET)
if(pybind11_FOUND)
  pybind11_add_module(tr122 bindings.cpp)
  target_link_libraries(tr122 PRIVATE Eigen3::Eigen)
  target_include_directories(tr122 PRIVATE src)
else()
  message(WARNING "pybind11 not found; building C++ tests only")
  add_executable(tr122_selftest tests/selftest.cpp)
  target_link_libraries(tr122_selftest PRIVATE Eigen3::Eigen)
  target_include_directories(tr122_selftest PRIVATE src)
endif()

enable_testing()
add_executable(tr122_selftest tests/selftest.cpp)
target_link_libraries(tr122_selftest PRIVATE Eigen3::Eigen)
target_include_directories(tr122_selftest PRIVATE src)
add_test(NAME selftest COMMAND tr122_selftest)
